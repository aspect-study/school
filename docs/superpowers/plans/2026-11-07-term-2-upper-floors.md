# Term 2 Upper Floors Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move Term 2 from the Study Loft above My Little House to a balcony floor on every subject building: the
ground door opens `&term=1`, side stairs lead to a balcony with a 🌟 Term 2 door (`&term=2`) and a 📝 exam board
(`&term=2&exam=1`).

**Architecture:** A new pure module `upstairs.js` holds every building's balcony, stairs, spots, URLs and the
guide's floor rule (tested under `node --test`). `upstairs3d.js` draws stairs and balconies; `build.js` makes each body
one floor taller. `world-main.js` gains an `up` state like its `home` state: a movement box, a lift height for her,
her pet and the camera, and three balcony spots. The loft, its ladder and the Old lessons bookcase are removed.

**Tech Stack:** ES5 IIFE browser modules on Three.js, `node --test`, headless-Chrome e2e (`tests/e2e`).

Spec: `docs/superpowers/specs/2026-11-07-term-2-upper-floors-design.md`.

**Rules for every task (from `CLAUDE.md`):** ES5 only in `web/` (no arrow functions, `let`, template strings). Comments
only for non-obvious code. Never run `git commit`: leave changes unstaged; the owner gets one commit command at the
end (Task 10). Read `docs/world-modules.md` before opening world files, and read `world-main.js` by line range.

- [ ] **Step 4: Write `web/world/upstairs.js`**

```js
/* Term 2 upstairs on every subject building: the balcony across its front (Term 2 door, exam board), the spiral stairs
   at its north front corner, the spots she uses, the three game links and which floor the guide's step belongs to.
   Pure (no Three.js). Distances "out" run from a building's centre toward the street. */
(function (root) {
  'use strict';
  var node = typeof module !== 'undefined' && module.exports;
  var Layout = node ? require('./layout.js') : root.World3D.Layout;
  var B = Layout.BUILDING;

  var FRONT = 4.0, RAIL = 6.8, MID = 5.4, HALF = 5;
  var STAIRS = { out: 5.2, north: 7.4, r: 1.4 }, FOOT_OUT = 7.8, FOOT_R = 1.8, POST = { out: 6.6, z: 4.8, r: 0.35 };
  var LAND_Z = -2.6, DOWN_Z = -4.1, EXAM_Z = 3.6;

  function countOf(terms, app) { return terms && terms[app] ? terms[app] : { lessons: 0, exams: 0, ids: [] }; }

  function info(b, terms) {
    var s = b.side, n = countOf(terms, b.app);
    var out = function (d) { return b.x - s * d; };
    return {
      app: b.app, folder: b.folder, emoji: b.emoji, sign: b.sign, roof: b.roof, side: s, y: B.upY,
      ready: n.lessons > 0, examReady: n.lessons + n.exams > 0,
      wall: out(FRONT), rail: out(RAIL),
      stairs: { x: out(STAIRS.out), z: b.z - STAIRS.north, r: STAIRS.r },
      foot: { x: out(FOOT_OUT), z: b.z - STAIRS.north, face: Math.atan2(-s, 0) },
      box: { minX: Math.min(out(FRONT), out(RAIL)), maxX: Math.max(out(FRONT), out(RAIL)), minZ: b.z - HALF, maxZ: b.z + HALF },
      head: { x: out(MID), z: b.z + LAND_Z, face: 0 },
      down: { x: out(MID), z: b.z + DOWN_Z },
      door: { x: out(MID), z: b.z },
      exam: { x: out(MID), z: b.z + EXAM_Z },
      posts: [{ x: out(POST.out), z: b.z - POST.z, r: POST.r }, { x: out(POST.out), z: b.z + POST.z, r: POST.r }]
    };
  }

  function all(grade, terms) { return Layout.buildings(grade).map(function (b) { return info(b, terms); }); }
  function find(grade, terms, app) { return all(grade, terms).filter(function (i) { return i.app === app; })[0] || null; }

  // Street-level: the Go up spot at each stairs' foot, and what she walks round (the stairs, the balcony posts).
  function feet(grade) {
    return all(grade, {}).map(function (i) { return { kind: 'stairs-up', id: 'up:' + i.app, app: i.app, x: i.foot.x, z: i.foot.z, r: FOOT_R }; });
  }
  function solids(grade) {
    var out = [];
    all(grade, {}).forEach(function (i) { out.push(i.stairs, i.posts[0], i.posts[1]); });
    return out;
  }

  // On the balcony: the three spots, and the railing round it.
  function items(i) {
    return [
      { kind: 'up-door', id: 'up-door:' + i.app, app: i.app, x: i.door.x, z: i.door.z, r: 1.4 },
      { kind: 'up-exam', id: 'up-exam:' + i.app, app: i.app, x: i.exam.x, z: i.exam.z, r: 1.4 },
      { kind: 'stairs-down', id: 'stairs-down:' + i.app, app: i.app, x: i.down.x, z: i.down.z, r: 1.0 }
    ];
  }
  function blocked(i, x, z, r) {
    var b = i.box;
    return x - r < b.minX || x + r > b.maxX || z - r < b.minZ || z + r > b.maxZ;
  }

  function lessonOf(snap) {
    var s = snap && snap.step;
    return s && snap.stepHref && typeof s.lesson === 'string' && s.lesson ? s.lesson : null;
  }
  function termOf(terms, app, id) { var ids = countOf(terms, app).ids || []; return ids.indexOf(id) >= 0 ? 2 : 1; }
  function stepFloor(snap, terms) {
    var id = lessonOf(snap);
    return id && termOf(terms, snap.step.app, id) === 2 ? 'upper' : 'ground';
  }
  // The guide's link with its lesson's term; review, boss, visit and mock steps keep the full game (both terms).
  function stepHref(snap, terms) {
    if (!snap || !snap.step || !snap.stepHref) return null;
    var id = lessonOf(snap);
    return id ? snap.stepHref + '&term=' + termOf(terms, snap.step.app, id) : snap.stepHref;
  }

  function links(grade, i, snap, terms) {
    var base = Layout.appUrl(grade, i.folder);
    var out = { ground: base + '&term=1', upper: base + '&term=2', exam: base + '&term=2&exam=1' };
    var href = snap && snap.step && snap.step.app === i.app ? stepHref(snap, terms) : null;
    if (href) out[stepFloor(snap, terms)] = href;
    return out;
  }
  function buddyHref(grade, i, snap, terms) {
    var l = links(grade, i, snap, terms);
    if (snap && snap.step && snap.step.app === i.app && snap.stepHref) return l[stepFloor(snap, terms)];
    return i.ready ? l.upper : l.ground;
  }

  var exported = { FOOT_R: FOOT_R, info: info, all: all, find: find, feet: feet, solids: solids, items: items, blocked: blocked,
    stepFloor: stepFloor, stepHref: stepHref, links: links, buddyHref: buddyHref };
  if (node) {
    module.exports = exported;
    return;
  }
  root.World3D = root.World3D || {};
  root.World3D.Upstairs = exported;
})(this);
```

```js
/* The upper floors in 3D: on every subject building the spiral stairs at the north front corner, the balcony with its
   railing and posts, the Term 2 door (🌟, or 🌱 and greyed until Term 2 lessons exist), the exam board, and the
   Term 1 / Term 2 plaques. Positions come from upstairs.js (world coordinates). */
(function (root) {
  'use strict';
  var STEPS = 12, TURN = Math.PI * 1.5;

  // o = { floors: Upstairs.all(...), term1, term2 } (English sign words for the plaques).
  function create(S, o) {
    var THREE = S.THREE, group = new THREE.Group(), myTex = [], mySprites = [], dim = [];
    S.scene.add(group);

    function add(geo, color, x, y, z) { return S.add(geo, color, x, y, z, group); }
    function sprite(tex, w, h, x, y, z) {
      myTex.push(tex);
      var s = S.sprite(tex, w, h);
      mySprites.push(s.material);
      s.position.set(x, y, z);
      group.add(s);
      return s;
    }
    function label(text, color, w, x, y, z) { return sprite(S.label(text, color), w, w * 180 / 512, x, y, z); }

    o.floors.forEach(function (i) {
      var y = i.y, s = i.side, midX = (i.wall + i.rail) / 2, depth = Math.abs(i.rail - i.wall), len = i.box.maxZ - i.box.minZ;
      var face = -s;
      add(S.rbox(depth + 0.2, 0.3, len, 0.1), '#fff6fb', midX, y - 0.15, (i.box.minZ + i.box.maxZ) / 2);
      add(S.rbox(0.2, 1.1, len, 0.06), '#ffffff', i.rail, y + 0.55, (i.box.minZ + i.box.maxZ) / 2);
      add(S.rbox(depth, 1.1, 0.2, 0.06), '#ffffff', midX, y + 0.55, i.box.maxZ);
      i.posts.forEach(function (p) { add(S.cyl(p.r, p.r, y, 10), '#ffffff', p.x, y / 2, p.z); });

      add(S.cyl(0.25, 0.25, y + 1.2, 10), '#c98b6b', i.stairs.x, (y + 1.2) / 2, i.stairs.z);
      for (var k = 0; k < STEPS; k++) {
        var a = (k / STEPS) * TURN, st = add(S.rbox(i.stairs.r, 0.18, 0.7, 0.05), '#ffd9a8',
          i.stairs.x + Math.cos(a) * i.stairs.r / 2, (k + 1) * y / STEPS, i.stairs.z + Math.sin(a) * i.stairs.r / 2);
        st.rotation.y = -a;
      }

      var ready = i.ready, doorX = i.wall + face * 0.3;
      add(S.rbox(0.5, 3.4, 2.4, 0.25), ready ? '#c98b6b' : '#d8d8e0', doorX, y + 1.7, i.door.z);
      sprite(S.emoji(ready ? '🌟' : '🌱'), 0.9, 0.9, doorX + face * 0.4, y + 3.9, i.door.z);
      label(o.term2, ready ? i.roof : '#b8b8c8', 3.2, doorX + face * 0.4, y + 4.9, i.door.z);
      add(S.rbox(0.25, 1.8, 2.2, 0.06), i.examReady ? '#5a7f5a' : '#b8b8c8', i.wall + face * 0.2, y + 1.9, i.exam.z);
      sprite(S.emoji('📝'), 0.8, 0.8, i.wall + face * 0.45, y + 2.0, i.exam.z);
      label(o.term1, i.roof, 3.2, i.wall + face * 0.6, 5.9, i.door.z + 2.6);
      if (!ready) dim.push(i.app);
    });

    return {
      group: group,
      count: function () { return o.floors.length; },
      dimmed: function () { return dim.slice(); },
      dispose: function () {
        S.scene.remove(group);
        mySprites.forEach(function (m) { m.dispose(); });
        myTex.forEach(function (t) { if (t && t.dispose) t.dispose(); });
      }
    };
  }

  var exported = { create: create };
  if (typeof module !== 'undefined' && module.exports) {
    module.exports = exported;
    return;
  }
  root.World3D = root.World3D || {};
  root.World3D.Upstairs3D = exported;
})(this);
```
