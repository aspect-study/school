# Pets + Pet Gear (Part 2a) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Mang Kiko's Pet Stall in Shop Plaza sells 13 more pets and 14 pet gear pieces for coins; the mirror gets a Pet
tab (pet, colour, name, where it rides, gear); her pet follows her, sits, sleeps, shows its name, hops with hearts when
tapped and munches a free treat; her sister's pet shows at the gate with its gear.

**Architecture:** Pure, Node-testable data and logic (`pets.js` catalog + `wearable()`, `look.js` pet fields,
`petlife.js` mood timers) feed a drawing layer (`petbody.js` pet and gear builders, mounts) and a runtime controller
(`companion.js`: follow, ride, name tag, bubble, tap, treat). The boutique panel becomes a shared `stall.js` with two thin
wrappers (`boutique.js`, `petshop.js`). Paid pets and gear go in the existing `wardrobe_v1` owned list; pet choices live
in `avatar_v1`.

**Tech Stack:** ES5 browser JS in IIFEs `(function (root) {...})(this)`, Three.js r149 (vendored), `node --test`,
headless Chrome e2e (`tests/e2e`). No build step.

**Spec:** `docs/superpowers/specs/2026-10-06-pets-design.md`

**Small changes from the spec (decided while planning):**
- A new runtime file `web/world/companion.js` holds the pet's life in the world, so `world-main.js` (already 500 lines)
  only wires it.
- `play.js` is not touched: during a playground ride a riding pet stays on her (it rides too), and a walking pet waits.
- The spec's `Pets.petLook()` is named `Pets.wearable()`, like `Items.wearable()`, and returns the whole look.
- The name tag is a child of the pet, so it shrinks with a riding pet.

      var g = find(w[s]);
      out.petWear[s] = g && g.kind === 'gear' && g.slot === s && has(owned, g) ? g.id : '';
    });
    return out;
  }

  // another device, waiting for wardrobe_v1 to sync, or from a newer version) and an empty slot for such gear.
  function keepUnowned(stored, edited, owned) {
    var out = Object.assign({}, edited), wear = Object.assign({}, edited.petWear), s = stored || {}, was = s.petWear || {};
    var sp = petOf(s.pet), hidden = typeof s.pet === 'string' && s.pet !== 'chick' && (!sp || !has(owned, sp));
    if (hidden && out.pet === 'chick') {
      out.pet = s.pet;
      out.petColor = s.petColor;
      out.petSpot = s.petSpot;
    }
    GEAR_SLOTS.forEach(function (k) {
      var g = find(was[k]);
      if (!wear[k] && was[k] && (!g || !has(owned, g))) wear[k] = was[k];
    });
    out.petWear = wear;
    return out;
  }

  var exported = {
    COLORS: COLORS, SPOTS: SPOTS, SIZES: SIZES, GEAR_SLOTS: GEAR_SLOTS, GROUPS: GROUPS, PETS: PETS, GEAR: GEAR,
    find: find, spotsFor: spotsFor, inGroup: inGroup, name: name, emptyWear: emptyWear, wearable: wearable, keepUnowned: keepUnowned
  };
  if (typeof module !== 'undefined' && module.exports) {
    module.exports = exported;
    return;
  }
  root.World3D = root.World3D || {};
  root.World3D.Pets = exported;
})(this);

```js
/* Her pet's mood, with no Three.js: it sits 3 s after she stops and sleeps after 10 s, wakes with a stretch when she
   moves, hops after a pat and munches a treat (free, at most one every 2 s). Nothing is saved; it never gets hungry. */
(function (root) {
  'use strict';

  var SIT = 3, SLEEP = 10, FAR = 12, TREAT_WAIT = 2, SISTER = 6, HAPPY = 1.2, MUNCH = 1.6, STRETCH = 0.6;

  function create() { return { t: 0, still: 0, happy: 0, munch: 0, stretch: 0, treatAt: -Infinity }; }

  // her = { moving }
  function tick(s, dt, her) {
    s.t += dt;
    s.happy = Math.max(0, s.happy - dt);
    s.munch = Math.max(0, s.munch - dt);
    s.stretch = Math.max(0, s.stretch - dt);
    if (her.moving) {
      if (s.still >= SLEEP) s.stretch = STRETCH;
      s.still = 0;
    } else s.still += dt;
    return s;
  }

  function pat(s) {
    s.happy = HAPPY;
    s.still = 0;
  }

  function treat(s) {
    if (s.t - s.treatAt < TREAT_WAIT) return false;
    s.treatAt = s.t;
    s.munch = MUNCH;
    s.still = 0;
    return true;
  }

  // moved: the pet itself moved this frame.
  function mood(s, moved) {
    if (s.happy > 0) return 'happy';
    if (s.munch > 0) return 'munch';
    if (s.stretch > 0) return 'stretch';
    if (s.still >= SLEEP) return 'sleep';
    if (s.still >= SIT) return 'sit';
    return moved ? 'walk' : 'idle';
  }

  // sisterD: how far her sister stands, or null when she is not here.
  function bubble(s, sisterD) {
    if (s.still >= SLEEP && s.happy <= 0 && s.munch <= 0) return '💤';
    if (s.munch > 0) return '🍪';
    if (s.happy > 0) return '💖';
    return sisterD !== null && sisterD !== undefined && sisterD <= SISTER ? '💖' : '';
  }

  function far(d) { return d >= FAR; }

  var exported = { SIT: SIT, SLEEP: SLEEP, FAR: FAR, TREAT_WAIT: TREAT_WAIT, SISTER: SISTER, create: create, tick: tick, pat: pat, treat: treat, mood: mood, bubble: bubble, far: far };
  if (typeof module !== 'undefined' && module.exports) {
    module.exports = exported;
    return;
  }
  root.World3D = root.World3D || {};
  root.World3D.PetLife = exported;
})(this);
```

```js
/* Her pet in the 3D world: drawn from her look (petbody.js), walking beside her or riding on her (shoulder, head or
   arms), with its name above it and a mood bubble. petlife.js decides the mood; a tap pets it and the 🍪 Treat button
   feeds it. world-main.js calls set() whenever it dresses her, place() when she jumps somewhere and tick() each frame. */
(function (root) {
  'use strict';
  var W = root.World3D = root.World3D || {};
  var SIDE = 1.4, BEHIND = 2.6, PET_R = 0.6;

  // o = { S, canvas, blocked(x, z, r), sparkle(x, z), hearts(x, y, z), sound(name) }
  function create(o) {
    var S = o.S, THREE = S.THREE, PB = W.PetBody, PL = W.PetLife, Walk = W.Walk;
    var life = PL.create(), pet = null, spot = 'walk', bubble = null, shown = '', mood = 'idle', tex = {};
    var ray = new THREE.Raycaster(), ndc = new THREE.Vector2(), v = new THREE.Vector3();

    function emoji(e) { return tex[e] || (tex[e] = S.emoji(e)); }

    // l: a look already through Pets.wearable(); me: her character.
    function set(l, me) {
      if (pet) {
        if (pet.group.parent) pet.group.parent.remove(pet.group);
        pet.dispose();
        bubble.material.dispose();
      }
      pet = PB.build(S, l);
      PB.tag(S, pet, W.Look.petName(l));
      bubble = S.sprite(emoji('💤'), 1.1, 1.1);
      bubble.visible = false;
      bubble.position.set(0, pet.top + 2, 0);
      pet.group.add(bubble);
      shown = '';
      spot = PB.mount(pet, me, l.petSpot) ? l.petSpot : 'walk';
      if (spot === 'walk') S.scene.add(pet.group);
    }

    function beside(her) {
      return {
        x: her.x - Math.sin(her.face) * BEHIND + Math.cos(her.face) * SIDE,
        z: her.z - Math.cos(her.face) * BEHIND - Math.sin(her.face) * SIDE
      };
    }
    function blockedPet(x, z) { return o.blocked(x, z, PET_R); }
    // Beside her if there is room, else where she stands (she always fits there).
    function free(her) {
      var b = beside(her);
      return blockedPet(b.x, b.z) ? { x: her.x, z: her.z } : b;
    }

    function place(her) {
      if (!pet || spot !== 'walk') return;
      var f = free(her);
      pet.group.position.set(f.x, 0, f.z);
    }

    // her = { x, z, face, moving, hold (a panel, chat or ride: the pet waits), sisterD }
    function tick(dt, t, her) {
      if (!pet) return;
      PL.tick(life, dt, her);
      var moved = false;
      if (spot === 'walk' && !her.hold) {
        var p = pet.group.position, b = beside(her);
        if (PL.far(Math.hypot(her.x - p.x, her.z - p.z))) {
          var f = free(her);
          p.set(f.x, 0, f.z);
          o.sparkle(f.x, f.z);
        } else {
          var dir = Walk.toward(p.x, p.z, b.x, b.z);
          if (dir) {
            dir.mag = Math.min(1.3, Math.hypot(b.x - p.x, b.z - p.z) / 1.5);
            var n = Walk.step(p.x, p.z, dir, dt, blockedPet);
            moved = n.x !== p.x || n.z !== p.z;
            p.x = n.x;
            p.z = n.z;
          }
        }
        if (Math.hypot(her.x - p.x, her.z - p.z) > 0.5) pet.group.lookAt(her.x, 0, her.z);
      }
      mood = PL.mood(life, moved);
      pet.animate(t, mood);
      var e = PL.bubble(life, her.sisterD);
      if (e !== shown) {
        shown = e;
        bubble.visible = !!e;
        if (e) {
          bubble.material.map = emoji(e);
          bubble.material.needsUpdate = true;
        }
      }
    }

    function pat() {
      if (!pet) return;
      PL.pat(life);
      pet.group.getWorldPosition(v);
      o.hearts(v.x, v.y + 1.5, v.z);
      o.sound('cardRead');
    }

    // A tap on the screen at (cx, cy): true (and a pat) when it lands on her pet.
    function hit(cx, cy) {
      if (!pet) return false;
      S.scene.updateMatrixWorld();
      S.camera.updateMatrixWorld();
      var r = o.canvas.getBoundingClientRect();
      ndc.set((cx - r.left) / r.width * 2 - 1, -((cy - r.top) / r.height) * 2 + 1);
      ray.setFromCamera(ndc, S.camera);
      if (!ray.intersectObject(pet.body, true).length) return false;
      pat();
      return true;
    }

    function treat() {
      if (!pet || !PL.treat(life)) return false;
      o.sound('allRead');
      return true;
    }

    // For the e2e driver: tap the middle of her pet on the screen.
    function tapCenter() {
      if (!pet) return false;
      S.scene.updateMatrixWorld();
      S.camera.updateMatrixWorld();
      var r = o.canvas.getBoundingClientRect();
      pet.group.localToWorld(v.set(0, pet.top * 0.35, 0));
      v.project(S.camera);
      return hit(r.left + (v.x + 1) / 2 * r.width, r.top + (1 - v.y) / 2 * r.height);
    }

    function state() {
      var p = pet && spot === 'walk' ? pet.group.position : null;
      return { mood: mood, bubble: shown, spot: spot, x: p ? p.x : null, z: p ? p.z : null };
    }

    return { set: set, place: place, tick: tick, hit: hit, pat: pat, treat: treat, tapCenter: tapCenter, state: state };
  }

  W.Companion = { create: create };
})(this);
```
